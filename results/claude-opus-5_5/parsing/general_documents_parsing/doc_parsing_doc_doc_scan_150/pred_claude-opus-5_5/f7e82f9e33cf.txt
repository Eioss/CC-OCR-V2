\toprule
name & formula \\
\midrule

multiplication by impulse & $f(n)\,\delta(n) = f(0)\,\delta(n)$ \\[1ex]
\ldots by shifted impulse & $f(n)\,\delta(n-n_o) = f(n_o)\,\delta(n-n_o)$ \\[1ex]

\noindent\textbf{formulas for continuous-time LTI signals and systems}

\medskip
\noindent
\begin{tabular}{@{}ll@{}}
\toprule
name & formula \\
\midrule
area under impulse & $\displaystyle\int \delta(t)\,dt = 1$ \\[2ex]
multiplication by impulse & $f(t)\,\delta(t) = f(0)\,\delta(t)$ \\[1ex]
\ldots by shifted impulse & $f(t)\,\delta(t-t_o) = f(t_o)\,\delta(t-t_o)$ \\[1ex]
convolution & $\displaystyle f(t) * g(t) = \int f(\tau)\,g(t-\tau)\,d\tau$ \\[2ex]
\ldots with an impulse & $f(t) * \delta(t) = f(t)$ \\[1ex]
\ldots with a shifted impulse & $f(t) * \delta(t-t_o) = f(t-t_o)$ \\[1ex]
transfer function & $\displaystyle H(s) = \int h(t)\,e^{-st}\,dt$ \\[2ex]
frequency response & $\displaystyle H^f(\omega) = \int h(t)\,e^{-j\omega t}\,dt$ \\[2ex]
\ldots their connection & $H^f(\omega) = H(j\omega)$ \\
 & \quad provided $j\omega$-axis $\subset$ ROC \\
\bottomrule
\end{tabular}
