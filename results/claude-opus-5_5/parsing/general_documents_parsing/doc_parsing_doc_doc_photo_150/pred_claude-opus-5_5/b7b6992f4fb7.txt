```latex
\documentclass[11pt]{article}

\begin{document}

\noindent\textbf{SEQUENCE AND SERIES} \hfill \textbf{8.3}

\begin{center}
\underline{\textbf{\Large OBJECTIVE QUESTIONS}}
\end{center}

\begin{multicols}{2}
\begin{enumerate}[label=\textbf{\arabic*.}, leftmargin=*]

% Question 1
\item If the sum of the first $2n$ terms of the A.P. $2, 5, 8, \ldots$ is equal to the sum of the first $n$ terms of the A.P. $57, 59, 61, \ldots$, then $n$ equals

\begin{tabular}{p{0.4\linewidth}p{0.4\linewidth}}
(\textit{a}) $10$ & (\textit{b}) $12$ \\
(\textit{c}) $11$ & (\textit{d}) $13$
\end{tabular}

% Question 2
\item If $x, y, z$ are three positive numbers in G.P. then
\[
\frac{1}{1+\ln x},\ \frac{1}{1+\ln y},\ \frac{1}{1+\ln z} \text{ are in}
\]
(\textit{a}) A.P.\\
(\textit{b}) G.P.\\
(\textit{c}) H.P.\\
(\textit{d}) none of these

% Question 3
\item Let $a, b, c$ are in A.P. and $a^2, b^2, c^2$ are in G.P. If $a < b < c$ and $a + b + c = \dfrac{3}{2}$, then value of $a$ is

\begin{tabular}{p{0.4\linewidth}p{0.4\linewidth}}
(\textit{a}) $\dfrac{1}{2\sqrt{2}}$ & (\textit{b}) $\dfrac{1}{2\sqrt{3}}$ \\[3mm]
(\textit{c}) $\dfrac{1}{2} - \dfrac{1}{\sqrt{3}}$ & (\textit{d}) $\dfrac{1}{2} - \dfrac{1}{\sqrt{2}}$
\end{tabular}

% Question 4
\item If $1, \log_y x, \log_z y, -15 \log_x z$ are in A.P., then

\begin{tabular}{p{0.4\linewidth}p{0.4\linewidth}}
(\textit{a}) $z^3 = x$ & (\textit{b}) $x = y^{-1}$ \\
(\textit{c}) $z^{-3} = y$ & (\textit{d}) all of these
\end{tabular}

% Question 5
\item The sum to $n$ terms of the series
\[
\frac{2}{3} + \frac{8}{9} + \frac{26}{27} + \frac{80}{81} + \ldots\ldots \text{ is equal to}
\]

\begin{tabular}{p{0.4\linewidth}p{0.4\linewidth}}
(\textit{a}) $1 - \left(\dfrac{1}{3}\right)^n$ & (\textit{b}) $2 - \dfrac{1}{2}\left(\dfrac{2}{3}\right)^n$ \\[3mm]
(\textit{c}) $n - n\left(\dfrac{1}{3}\right)^n$ & (\textit{d}) none of these
\end{tabular}

% Question 6
\item Let $a_n$ be the $n$th term of a G.P. of positive integers. Let $\displaystyle\sum_{n=1}^{100} a_{2n} = \alpha$ and $\displaystyle\sum_{n=1}^{100} a_{2n+1} = \beta$ such that $a \neq \beta$. Thus common ratio of the G.P. is

\begin{tabular}{p{0.4\linewidth}p{0.4\linewidth}}
(\textit{a}) $\dfrac{\alpha}{\beta}$ & (\textit{b}) $\dfrac{\beta}{\alpha}$ \\[3mm]
(\textit{c}) $\left(\dfrac{\alpha}{\beta}\right)^{1/2}$ & (\textit{d}) $\left(\dfrac{\beta}{\alpha}\right)^{1/2}$
\end{tabular}

% Question 7
\item Let $T_r$ be the $r$th term of an A.P. for $r = 1, 2, 3 \ldots$ If for some positive integers $m, n$
\[
T_m = \frac{1}{n},\ T_n = \frac{1}{m}, \text{ then } T_{mn} \text{ equals}
\]

\begin{tabular}{p{0.4\linewidth}p{0.4\linewidth}}
(\textit{a}) $\dfrac{1}{mn}$ & (\textit{b}) $\dfrac{1}{m} + \dfrac{1}{n}$ \\[3mm]
(\textit{c}) $1$ & (\textit{d}) $0$
\end{tabular}

% Question 8
\item If $a_1, a_2, \ldots, a_n$ are positive real numbers whose product is a fixed number $c$, then the minimum value of $a_1 + a_2 + \ldots\ldots a_{n-1} + 2a_n$ is

(\textit{a}) $n(2c)^{1/n}$\\
(\textit{b}) $(n+1)c^{1/n}$\\
(\textit{c}) $2nc^{1/n}$\\
(\textit{d}) $(n+1)(2c)^{1/n}$

% Question 9
\item If the first term of an A.P. of consecutive integers is $p^2 + 1$, then sum of $(2p+1)$ terms of this series can be expressed as

\begin{tabular}{p{0.4\linewidth}p{0.4\linewidth}}
(\textit{a}) $(p+1)^2$ & (\textit{b}) $(2p+1)(p+1)^2$ \\
(\textit{c}) $(p+1)^3$ & (\textit{d}) $p^3 + (p+1)^3$
\end{tabular}

% Question 10
\item If $x > 1$ and $\left(\dfrac{1}{x}\right)^a, \left(\dfrac{1}{x}\right)^b, \left(\dfrac{1}{x}\right)^c$ are in G.P., then $a, b, c$ are in

\begin{tabular}{p{0.4\linewidth}p{0.4\linewidth}}
(\textit{a}) A.P. & (\textit{b}) G.P. \\
(\textit{c}) H.P. & (\textit{d}) none of these
\end{tabular}

% Question 11
\item If the positive numbers $a, b, c, d$ be in A.P, then $abc, abd, bcd$ are

(\textit{a}) not in A.P., G.P. and H.P.\\
(\textit{b}) in A.P.\\
(\textit{c}) in G.P.\\
(\textit{d}) in H.P.

% Question 12
\item The sixth term of an A.P. is $2$ and its common difference is greater than one. The value of the common difference of the progression so that the product of the first, fourth and fifth terms is greatest, is

\begin{tabular}{p{0.4\linewidth}p{0.4\linewidth}}
(\textit{a}) $\dfrac{8}{5}$ & (\textit{b}) $\dfrac{2}{3}$ \\[3mm]
(\textit{c}) $\dfrac{5}{8}$ & (\textit{d}) $\dfrac{3}{2}$
\end{tabular}

\end{enumerate}
\end{multicols}
```
