\[
\bullet\quad \sin(A+B)=\Sin A\cdot \cos B+\cos A\cdot \sin B
\]

\[

\[

\[
\bullet\quad \cot(A+B)=\frac{\cot B\cdot \cot A-1}{\cot B+\cot A}

\[
\bullet\quad \cot(A-B)=\frac{\cot B\cdot \cot A+1}{\cot B-\cot A}

\[
*\text{\underline{General Solution of trigonometry}}
\]

\[
\bullet\quad \sin x=0\Rightarrow x=n\pi
\]

\[
\bullet\quad \cos x=0\Rightarrow x=(2n+1)\frac{\pi}{2}
\]

\[
\bullet\quad \tan x=0\Rightarrow x=n\pi
\]

\[
\bullet\quad \sin x=\sin y\Rightarrow x=n\pi+(-1)^n\cdot y
\]

\[
\bullet\quad \cos x=\cos y\Rightarrow x=2n\pi\pm y
\]

\[
\bullet\quad \tan x=\tan y\Rightarrow x=n\pi+y
\]

\[
\left.
\begin{array}{l}
\bullet\quad \sin^2x=\sin^2y\\
\bullet\quad \cos^2x=\cos^2y\\
\bullet\quad \tan^2x=\tan^2y
\end{array}
\right\}
\Rightarrow x=n\pi\pm y
\]

\[
\boxed{n\in Z}
