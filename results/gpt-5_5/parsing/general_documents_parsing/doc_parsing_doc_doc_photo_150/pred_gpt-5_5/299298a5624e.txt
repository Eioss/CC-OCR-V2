\section*{Trigonometry}

\subsection*{1) Trigonometry Ratios}

\[
\operatorname{Sin}\theta=\frac{\operatorname{Opp}}{\operatorname{Hyp}}
\qquad
\operatorname{Cos}\theta=\frac{\operatorname{Adj}}{\operatorname{Hyp}}
\]

\[
\tan\theta=\frac{\operatorname{Opp}}{\operatorname{Adj}}
\qquad
\operatorname{Cot}\theta=\frac{\operatorname{Adj}}{\operatorname{Opp}}
\]

\[
\operatorname{Sec}\theta=\frac{\operatorname{Hyp}}{\operatorname{Adj}}
\qquad
\operatorname{Cosec}\theta=\frac{\operatorname{Hyp}}{\operatorname{Opp}}
\]

\subsection*{2) Imp - Results}

\[
\begin{array}{ll}
\text{i)} & \operatorname{Sin}\theta \cdot \operatorname{Cosec}\theta=1 \\[2mm]
\text{ii)} & \operatorname{Cos}\theta \cdot \operatorname{Sec}\theta=1 \\[2mm]
\text{iii)} & \tan\theta \cdot \operatorname{Cot}\theta=1 \\[2mm]
\text{iv)} & \tan\theta=\dfrac{\operatorname{Sin}\theta}{\operatorname{Cos}\theta} \\[3mm]
\text{v)} & \operatorname{Cot}\theta=\dfrac{\operatorname{Cos}\theta}{\operatorname{Sin}\theta} \\[3mm]
\text{vi)} & \operatorname{Sin}^{2}\theta+\operatorname{Cos}^{2}\theta=1 \\[2mm]
\text{vii)} & \operatorname{Sec}^{2}\theta=1+\tan^{2}\theta \\[2mm]
\text{viii)} & \operatorname{Cosec}^{2}\theta=1+\operatorname{Cot}^{2}\theta \\[2mm]
\text{ix)} & \pi^{c}=180^{\circ} \\[2mm]
\text{x)} & \text{length of arc } s=r\theta \\[2mm]
\text{xi)} & \text{Area of sector }=\dfrac{1}{2}r^{2}\theta
\end{array}
\]

\subsection*{4) Allied - Angles}

\[
\begin{array}{ll}
\text{1)} & \operatorname{Sin}(-\theta)=-\operatorname{Sin}\theta \\
& \operatorname{Cos}(-\theta)=\operatorname{Cos}\theta \\
& \tan(-\theta)=-\tan\theta \\
& \operatorname{Cot}(-\theta)=-\operatorname{Cot}\theta \\
& \operatorname{Sec}(-\theta)=\operatorname{Sec}\theta \\
& \operatorname{Cosec}(-\theta)=-\operatorname{Cosec}\theta \\[3mm]
\text{2)} & \operatorname{Sin}(90^{\circ}-\theta)=\operatorname{Cos}\theta \\
& \operatorname{Cos}(90^{\circ}-\theta)=\operatorname{Sin}\theta \\
& \tan(90^{\circ}-\theta)=\operatorname{Cot}\theta \\
& \operatorname{Cot}(90^{\circ}-\theta)=\tan\theta \\
& \operatorname{Sec}(90^{\circ}-\theta)=\operatorname{Cosec}\theta \\
& \operatorname{Cosec}(90^{\circ}-\theta)=\sec\theta
\end{array}
\]

\subsection*{5) Compound Angles}

\[
\begin{array}{ll}
\text{i)} & \operatorname{Sin}(A+B)=\operatorname{Sin}A\operatorname{Cos}B+\operatorname{Cos}A\operatorname{Sin}B \\[2mm]
\text{ii)} & \operatorname{Sin}(A-B)=\operatorname{Sin}A\operatorname{Cos}B-\operatorname{Cos}A\operatorname{Sin}B \\[2mm]
\text{iii)} & \operatorname{Cos}(A+B)=\operatorname{Cos}A\operatorname{Cos}B-\operatorname{Sin}A\operatorname{Sin}B \\[2mm]
\text{iv)} & \operatorname{Cos}(A-B)=\operatorname{Cos}A\operatorname{Cos}B+\operatorname{Sin}A\operatorname{Sin}B \\[2mm]
\text{v)} & \tan(A+B)=\dfrac{\tan A+\tan B}{1-\tan A\tan B} \\[4mm]
\text{vi)} & \tan(A-B)=\dfrac{\tan A-\tan B}{1+\tan A\tan B}
\end{array}
\]

\subsection*{6) Multiple Angle and Sub Multiple Angle}

\[
\begin{array}{ll}
\text{i)} & \operatorname{Sin}2A=2\operatorname{Sin}A\operatorname{Cos}A \\[2mm]
\text{ii)} & \operatorname{Cos}2A=\operatorname{Cos}^{2}A-\operatorname{Sin}^{2}A \\[2mm]
\text{iii)} & \tan 2A=\dfrac{2\tan A}{1-\tan^{2}A}
\end{array}
\]

\[
\begin{array}{ll}
\text{i)} & \operatorname{Sin}A=2\operatorname{Sin}\left(\dfrac{A}{2}\right)\operatorname{Cos}\left(\dfrac{A}{2}\right) \\[3mm]
\text{ii)} & \operatorname{Cos}A=\operatorname{Cos}^{2}\left(\dfrac{A}{2}\right)-\operatorname{Sin}^{2}\left(\dfrac{A}{2}\right) \\[3mm]
\text{iii)} & \operatorname{Cos}A=1-2\operatorname{Sin}^{2}\left(\dfrac{A}{2}\right) \\[3mm]
\text{iv)} & \operatorname{Cos}A=2\operatorname{Cos}^{2}\left(\dfrac{A}{2}\right)-1 \\[3mm]
\text{v)} & \operatorname{Cos}A=\dfrac{1-\tan^{2}\left(\dfrac{A}{2}\right)}{1+\tan^{2}\left(\dfrac{A}{2}\right)} \\[4mm]
\text{vi)} & \operatorname{Sin}A=\dfrac{2\tan\left(\dfrac{A}{2}\right)}{1+\tan^{2}\left(\dfrac{A}{2}\right)} \\[4mm]
\text{vii)} & \tan A=\dfrac{2\tan\left(\dfrac{A}{2}\right)}{1-\tan^{2}\left(\dfrac{A}{2}\right)} \\[4mm]
\text{viii)} & 1-\operatorname{Cos}A=2\operatorname{Sin}^{2}\left(\dfrac{A}{2}\right)
\end{array}
\]

\subsection*{7) Triple Angle Formulae}

\[
\begin{array}{ll}
\text{i)} & \sin 3\theta=3\sin\theta-4\sin^{3}\theta \\[2mm]
\text{ii)} & \cos 3\theta=4\cos^{3}\theta-3\cos\theta \\[2mm]
\text{iii)} & \tan 3\theta=\dfrac{3\tan\theta-\tan^{3}\theta}{1-3\tan^{2}\theta} \\[4mm]
\text{iv)} & \cos^{3}\theta=\dfrac{\cos 3\theta+3\cos\theta}{4} \\[4mm]
\text{v)} & \sin^{3}\theta=\dfrac{3\sin\theta-\sin 3\theta}{4}
\end{array}
\]

\subsection*{8) Factorization}

\[
\begin{array}{ll}
\text{i.} & \sin C+\sin D=2\sin\left(\dfrac{C+D}{2}\right)\cos\left(\dfrac{C-D}{2}\right) \\[4mm]
\text{ii.} & \sin C-\sin D=2\sin\left(\dfrac{C+D}{2}\right)\sin\left(\dfrac{C-D}{2}\right) \\[4mm]
\text{iii.} & \cos C+\cos D=2\cos\left(\dfrac{C+D}{2}\right)\cos\left(\dfrac{C-D}{2}\right) \\[4mm]
\text{iv.} & \cos C-\cos D=2\sin\left(\dfrac{C+D}{2}\right)\sin\left(\dfrac{D-C}{2}\right)\quad \text{if } C>D \\[4mm]
& =2\sin\left(\dfrac{C+D}{2}\right)\sin\left(\dfrac{D-C}{2}\right)\quad \text{if } D>C
\end{array}
\]

\subsection*{9) Defactorization formulae}

\[
\begin{array}{ll}
\text{i.} & 2\sin A\cos B=\sin(A+B)+\sin(A-B) \\[2mm]
\text{ii.} & 2\cos A\sin B=\sin(A+B)-\sin(A-B) \\[2mm]
\text{iii.} & 2\cos A\cos B=\cos(A+B)+\cos(A-B) \\[2mm]
\text{iv.} & 2\sin A\sin B=\cos(A-B)-\cos(A+B)
\end{array}
\]
