\[
\text{The steps to derive the quadratic formula are shown below.}
\]

\[
\begin{array}{|c|l|}
\hline
\text{Step 1} & ax^2 + bx + c = 0 \\
\hline
\text{Step 2} & ax^2 + bx = -c \\
\hline
\text{Step 3} & x^2 + \frac{b}{a}x = -\frac{c}{a} \\
\hline
\text{Step 4} & x^2 + \frac{b}{a}x + \frac{b^2}{4a^2}
= -\frac{c}{a} + \frac{b^2}{4a^2} \\
\hline
\text{Step 5} & x^2 + \frac{b}{a}x + \frac{b^2}{4a^2}
= -\frac{4ac}{4a^2} + \frac{b^2}{4a^2} \\
\hline
\text{Step 6} & \left(x + \frac{b}{2a}\right)^2
= \frac{b^2 - 4ac}{4a^2} \\
\hline
\text{Step 7} & \left(x + \frac{b}{2a}\right)
= \pm \sqrt{\frac{b^2 - 4ac}{4a^2}}
= \pm \frac{\sqrt{b^2 - 4ac}}{4a} \\
\hline
\text{Step 8} & x = -\frac{b}{2a} \pm \frac{\sqrt{b^2 - 4ac}}{2a} \\
\hline
\text{Step 9} & x = \frac{-b \pm \sqrt{b^2 - 4ac}}{2a} \\
\hline
\end{array}
\]

\[
\text{Which of the following is the first incorrect step?}
\]

\[
\begin{array}{l}
\circ\ \text{Step 4} \\
\circ\ \text{Step 6} \\
\circ\ \text{Step 7} \\
\circ\ \text{Step 8}
\end{array}
\]
