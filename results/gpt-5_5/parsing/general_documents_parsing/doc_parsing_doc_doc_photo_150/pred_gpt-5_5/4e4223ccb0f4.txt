\[
\begin{aligned}
\text{2. Find the LCM and HCF of the following pairs of} \\
\text{integers and verify that LCM } \times \text{ HCF = Product of the} \\
\text{two numbers}
\end{aligned}
\]

\[
\begin{aligned}
\text{(i) } & 26 \text{ and } 91 \\
& \text{Prime factors of } 26 = 2 \times 13 \\
& \text{Prime factors of } 91 = 7 \times 13 \\
& \text{HCF of } 26 \text{ and } 91 = 13 \\
& \text{LCM of } 26 \text{ and } 91 = 2 \times 7 \times 13 \\
& = 182
\end{aligned}
\]

\[
\begin{aligned}
& \text{Product of two numbers } = 26 \times 91 = 2366 \\
& \text{LCM } \times \text{ HCF } = 182 \times 13 = 2366
\end{aligned}
\]

\[
\text{So, Product of two numbers = LCM } \times \text{ HCF}
\]
