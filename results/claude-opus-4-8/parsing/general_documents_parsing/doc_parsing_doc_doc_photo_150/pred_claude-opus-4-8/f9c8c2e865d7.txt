\noindent\textbf{Additional examples:}

\begin{itemize}
    \item buffalo $\rightarrow$ buffaloes
    \item man $\rightarrow$ men
    \item woman $\rightarrow$ women
    \item child $\rightarrow$ children
    \item leaf $\rightarrow$ leaves
    \item thief $\rightarrow$ thieves
    \item knife $\rightarrow$ knives
    \item wife $\rightarrow$ wives
    \item life $\rightarrow$ lives
    \item man-servant $\rightarrow$ men-servants
\end{itemize}
