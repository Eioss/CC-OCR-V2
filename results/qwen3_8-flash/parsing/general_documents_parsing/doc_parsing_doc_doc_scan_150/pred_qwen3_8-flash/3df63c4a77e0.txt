    \large{\textbf{Mechanics}}
    \hrulefill
    \normalsize
    
    $x = x_0 + v_{x0}t + \frac{1}{2}a_x t^2$ \hfill $a_c = \frac{v^2}{r}$ \hfill $|\vec{F}_{spring}| = k|\vec{x}|$ \\[1em]
    $v = v_0 + at$ \hfill $\theta = \theta_0 + \omega_0 t + \frac{1}{2}\alpha t^2$ \hfill $PE_{spring} = \frac{1}{2}kx^2$ \\[1em]
    $v_x^2 - v_{x0}^2 = 2a(x-x_0)$ \hfill $\omega = \omega_0 + \alpha t$ \hfill $T_{spring} = 2\pi\sqrt{\frac{m}{k}}$ \\[1em]
    $\vec{a} = \frac{\Sigma \vec{F}}{m} = \frac{\vec{F}_{net}}{m}$ \hfill $T = \frac{2\pi}{\omega} = \frac{1}{f}$ \hfill $T_{pendulum} = 2\pi\sqrt{\frac{l}{g}}$ \\[1em]
    $|\vec{F}_{friction}| \leq \mu |\vec{F}_{Normal}|$ \hfill $v = f\lambda$ \\[1em]
    $\vec{p} = m\vec{v}$ \hfill $x = A\cos(2\pi ft)$ \hfill $|\vec{F}_{gravity}| = G\frac{m_1 m_2}{r^2}$ \\[1em]
    $\Delta\vec{p} = \vec{F}\Delta t$ \hfill $\vec{\alpha} = \frac{\Sigma \vec{\tau}}{I} = \frac{\vec{\tau}_{net}}{I}$ \hfill $|\vec{F}_{gravity}| = m\vec{g}$ \\[1em]
    $KE = \frac{1}{2}mv^2$ \hfill $\vec{\tau} = r \times F$ \hfill $PE_{gravity} = -G\frac{m_1 m_2}{r}$ \\[1em]
    $\Delta PE = mg\Delta y$ \hfill $L = I\omega$ \hfill $\rho = \frac{m}{V}$ \\[1em]
    $\Delta E = W = Fd\cos\theta$ \hfill $\Delta L = \tau\Delta t$ \hfill $KE = \frac{1}{2}I\omega^2$ \\[1em]

    $I = \frac{\Delta q}{\Delta t}$ \hfill $P = I\Delta V$ \\[1em]
    $R_{series} = R_1 + R_2 + ... + R_n$ \hfill $\frac{1}{R_{parallel}} = \frac{1}{R_1} + \frac{1}{R_2} + ... + \frac{1}{R_n}$ \\[1em]
