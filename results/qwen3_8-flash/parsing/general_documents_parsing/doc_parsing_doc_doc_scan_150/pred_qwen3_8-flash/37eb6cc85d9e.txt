\noindent
\begin{minipage}[t]{0.48\textwidth}
    \vspace{0pt}
    \textbf{Mean}\\[1ex]
    $\overline{x} = \frac{1}{n}\sum_{i=1}^{n}x_i$\\[2ex]
    
    \textbf{Standard Deviation*}\\[1ex]
    $S=\sqrt{\frac{\sum(x_i-\overline{x})^2}{n-1}}$\\[2ex]
    
    \textbf{Standard Error of the Mean*}\\[1ex]
    $SE_{\overline{x}}=\frac{S}{\sqrt{n}}$\\[2ex]
    
    \textbf{Chi-Square}\\[1ex]
    $\chi^2 = \sum \frac{(o-e)^2}{e}$\\[2ex]
    
    \begin{center}
        \textbf{Chi-Square Table}\\[1ex]
        \renewcommand{\arraystretch}{1.2}

    \vspace{0pt}
    $\overline{x}$ = sample mean\\[1ex]
    $n$ = size of the sample\\[1ex]
    $s$ = sample standard deviation (i.e., the sample-based estimate of the standard deviation of the population)\\[1ex]
    $o$ = observed results\\[1ex]
    $e$ = expected results\\[3ex]
    
    Degrees of freedom are equal to the number of distinct possible outcomes minus one.
\end{minipage}

\begin{minipage}[t]{0.48\textwidth}
    \vspace{0pt}
    \textbf{Laws of Probability}\\[1ex]
    If A and B are mutually exclusive, then:\\
    \hspace*{2em}$P(\text{A or B}) = P(\text{A}) + P(\text{B})$\\[1ex]
    If A and B are independent, then:\\
    \hspace*{2em}$P(\text{A and B}) = P(\text{A}) \times P(\text{B})$\\[2ex]
    
    \textbf{Hardy-Weinberg Equations}\\[1ex]
    $p^2 + 2pq + q^2 = 1$ \hspace{1em} $p$ = frequency of the dominant allele in a population\\[1ex]
    $p + q = 1$ \hspace{3.5em} $q$ = frequency of the recessive allele in a population
\end{minipage}%

\noindent
Mode = value that occurs most frequently in a data set\\[1ex]
Median = middle value that separates the greater and lesser halves of a data set\\[1ex]
Mean = sum of all data points divided by number of data points\\[1ex]
Range = value obtained by subtracting the smallest observation (sample minimum) from the greatest (sample maximum)
