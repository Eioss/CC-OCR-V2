\[
\sum_{n=1}^{\infty}\frac{1}{n^{2}+2n+1}
=\sum_{n=1}^{\infty}\frac{1}{(n+1)^{2}}
=\sum_{n=2}^{\infty}\frac{1}{n^{2}}
=\frac{\pi^{2}}{6}-1
\]

\[
\sum_{n=1}^{\infty}\frac{1}{n^{2}+4n+3}
=\sum_{n=1}^{\infty}\frac{1}{(n+1)(n+3)}
=\frac{1}{2}\sum_{n=1}^{\infty}\left(\frac{1}{n+1}-\frac{1}{n+3}\right)
=\frac{1}{2}\left(\frac{1}{2}+\frac{1}{3}\right)
=\frac{5}{12}
\]

\[
\sum_{n=1}^{\infty}\frac{1}{n^{2}+4n+4}
=\sum_{n=1}^{\infty}\frac{1}{(n+2)^{2}}
=\sum_{n=3}^{\infty}\frac{1}{n^{2}}
=\frac{\pi^{2}}{6}-1-\frac{1}{4}
=\frac{\pi^{2}}{6}-\frac{5}{4}
\]

\[
\sum_{n=1}^{\infty}\frac{1}{n^{2}+3n+2}
=\sum_{n=1}^{\infty}\frac{1}{(n+1)(n+2)}
=\sum_{n=1}^{\infty}\left(\frac{1}{n+1}-\frac{1}{n+2}\right)
=\frac{1}{2}
\]

\[
\sum_{n=1}^{\infty}\frac{1}{n^{2}+2n}
=\sum_{n=1}^{\infty}\frac{1}{n(n+2)}
=\frac{1}{2}\sum_{n=1}^{\infty}\left(\frac{1}{n}-\frac{1}{n+2}\right)
=\frac{1}{2}\left(1+\frac{1}{2}\right)
=\frac{3}{4}
\]
