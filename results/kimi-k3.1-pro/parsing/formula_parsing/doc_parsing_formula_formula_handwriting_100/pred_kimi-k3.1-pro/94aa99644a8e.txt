\begin{align*}
\|TX^{R}\|_{L_{\omega}^{r}L_{[0,T]}^{\infty}L_{x}^{2}}
&\leq \|X_{0}\|_{L_{\omega}^{r}L_{x}^{2}}
+\left\|\int_{0}^{t}S_{\beta}(t,s)\theta_{R}(X^{R})(s)
|X^{R}(s)|^{2\sigma}X^{R}(s)\,ds\right\|_{L_{\omega}^{r}L_{[0,T]}^{\infty}L_{x}^{2}}\\
&\leq \|X_{0}\|_{L_{\omega}^{r}L_{x}^{2}}
+C_{r,p}T^{\bar{\beta}}
\left\|\theta_{R}(X^{R})|X^{R}|^{2\sigma+1}\right\|_{L_{\omega}^{r}L_{[0,T]}^{r'}L_{x}^{p'}}\\
&=\|X_{0}\|_{L_{\omega}^{r}L_{x}^{2}}
+C_{r,p}T^{\bar{\beta}}
\left\|\left[\int_{0}^{T}\theta_{R}(X^{R})^{r'}(t)
\left(\int_{\mathbb{R}}|X^{R}(t)|^{(2\sigma+1)p'}\,dx\right)^{r'/p'}dt\right]^{1/r'}\right\|_{L_{\omega}^{r}}\\
\intertext{Since $p'=\dfrac{2\sigma+2}{2\sigma+1}$, the right hand side equals}
&=\|X_{0}\|_{L_{\omega}^{r}L_{x}^{2}}
+C_{r,p}T^{\bar{\beta}}
\left\|\left(\int_{0}^{T}\theta_{R}(X^{R})^{r'}(t)
\|X^{R}(t)\|_{L_{x}^{p}}^{(2\sigma+1)r'}\,dt\right)^{1/r'}\right\|_{L_{\omega}^{r}}\\
\intertext{Since $r\geq p=2\sigma+2$, we can apply H\"older's inequality to get}
&\leq \|X_{0}\|_{L_{\omega}^{r}L_{x}^{2}}
+C_{r,p}T^{\bar{\beta}}
\left\|\left(\int_{0}^{T}[\theta_{R}(X^{R})(t)]^{r/(2\sigma+1)}
\|X^{R}(t)\|_{L_{x}^{p}}^{r}\,dt\right)^{(2\sigma+1)/r}
T^{\frac{r-r'(2\sigma+1)}{rr'}}\right\|_{L_{\omega}^{r}}\\
\intertext{Letting $\widetilde{t}(\omega):=\inf\left\{s:\|X^{R}(\omega)\|_{L^{r}([0,s];L_{x}^{p})}\geq2R\right\}$, from definition of $\theta_{R}$ in (??), we have}
&=\|X_{0}\|_{L_{\omega}^{r}L_{x}^{2}}
+C_{r,p}T^{\widetilde{\beta}}
\left\|\left(\int_{0}^{\widetilde{t}}[\theta_{R}(X^{R})(t)]^{r/(2\sigma+1)}
\|X^{R}(t)\|_{L_{x}^{p}}^{r}\,dt\right)^{(2\sigma+1)/r}\right\|_{L_{\omega}^{r}}\\
\intertext{where $\widetilde{\beta}=\bar{\beta}+\dfrac{r-r'(2\sigma+1)}{rr'}>0$. Furthermore,}
&\leq \|X_{0}\|_{L_{\omega}^{r}L_{x}^{2}}
+C_{r,p}T^{\widetilde{\beta}}
\left\|\left(\int_{0}^{\widetilde{t}}\|X^{R}(t)\|_{L_{x}^{p}}^{r}\,dt\right)^{(2\sigma+1)/r}\right\|_{L_{\omega}^{r}}\\
&\leq \|X_{0}\|_{L_{\omega}^{r}L_{x}^{2}}
+C_{r,p}T^{\widetilde{\beta}}(2R)^{2\sigma}
\|X^{R}\|_{L_{\omega}^{r}L_{[0,T]}^{r}L_{x}^{p}}.
\end{align*}
