\section*{3\quad PRESENT CONTINUOUS}

\begin{tabular}{|>{\bfseries}p{0.18\textwidth}|p{0.72\textwidth}|}
\hline
Form & am, is, or are + verb + ing \\
\hline
Use & actions happening now, at the moment of\newline speaking. \\
\hline
Signal words & now, at the moment, still, Look! Listen!... \\
\hline
\end{tabular}

\exercise{1}{Find the signal words for present continuous.}

\begin{enumerate}[leftmargin=*,label=\arabic*.]
  \item Which is a signal word for present continuous?\\
  last -- now -- every night

  \item Which is \textbf{not} a signal word for present continuous?\\
  now - already - still
\end{enumerate}

\section*{4\quad PAST CONTINUOUS}

\begin{tabular}{|>{\bfseries}p{0.18\textwidth}|p{0.72\textwidth}|}
\hline
Form & was or were + verb + ing \\
\hline
Use & For actions which were in progress in the\newline past. \\
\hline
Signal words & when, while + long action in the past \\
\hline
\end{tabular}

\exercise{1}{Find the signal words for past continuous.}

  \item Which is a signal word for past continuous?\\

\section*{5\quad PRESENT PERFECT}

\begin{tabular}{|>{\bfseries}p{0.18\textwidth}|p{0.72\textwidth}|}
\hline
Form & have or has + Past Participle \\
\hline
Use & For actions which started in the past and\newline continued up to the present. \\
\hline
Signal words & just, already, yet, for, since, so far, recently,\newline ever... \\
\hline
\end{tabular}

\vspace{0.8em}

\noindent NB. We use Present Perfect Continuous (have/has+ been + verb +ing) to give
more emphasis on the duration of the action. However, some verbs can't be
used in the continuous form such as own, smell, need, belong, cost, agree,
contain..., etc.

\exercise{1}{Find the signal words for present perfect.}

\begin{enumerate}[leftmargin=*,label=\arabic*.]
  \item Which is a signal word for present perfect?\\
  still -- usually -- since

  \item Which is \textbf{not} a signal word for present perfect?\\
  since -- before -- for
\end{enumerate}

\exercise{2}{Put these verbs in present perfect.}

\begin{enumerate}[leftmargin=*,label=\arabic*.]
  \item They/ to do \hfill \answerline
  \item Jack / to publish \hfill \answerline
  \item He/ not, to have \hfill \answerline
  \item She/ to study \hfill \answerline
  \item Mike and Lina / to have \hfill \answerline
\end{enumerate}

\exercise{3}{Put the following verbs in the correct tense.}

\begin{enumerate}[leftmargin=*,label=\arabic*.]
  \item He (just, phone) Sabrina; she is in \hfill \answerline\\
  Chicago now.

  \item The maid (not, cook) the dinner yet.

  \item She (drive) this car for two hours.

  \item He (to be) in prison since 2003.

  \item (you, ever, read) Hemingway's novels?\\
  \answerline\\
  \answerline
\end{enumerate}
