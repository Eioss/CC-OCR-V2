\noindent {\large \textbf{2.5\quad Finance formulas -- Practice exercises}}

\hrule
\bigskip

\noindent \textsc{Compound Interest Formula:} \quad $a = p\left(1 + \frac{r}{12}\right)^{12y}$

\bigskip

\noindent \textsc{Equivalent APR Formula:} \quad $\mathrm{APR} = \left(1 + \frac{r}{12}\right)^{12} - 1$

\bigskip

\noindent \textsc{Future Value Annuity Formula:} \quad $a = p * \dfrac{\left(1 + \frac{r}{12}\right)^{12y} - 1}{\frac{r}{12}}$

\bigskip

\noindent \textsc{Loan Payment Formula:} \quad $p = \dfrac{a * \frac{r}{12}}{1 - \left(1 + \frac{r}{12}\right)^{-12y}}$

\bigskip

\noindent where

\hspace{2em} $a$ = account balance or loan amount (\$)

\hspace{2em} $p$ = initial deposit (principal), regular deposit, or regular payment (\$)

\hspace{2em} $y$ = time invested (years)

\hspace{2em} $r$ = interest rate compounded monthly (as a decimal)

\bigskip
\hrule
