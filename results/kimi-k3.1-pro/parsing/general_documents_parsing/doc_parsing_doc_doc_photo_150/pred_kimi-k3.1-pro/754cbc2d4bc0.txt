\section*{TRIGONOMETRY}

\begin{itemize}
    \item \(\sin(A+B)=\sin A\cdot\cos B+\cos A\cdot\sin B\)
    \item \(\sin(A-B)=\sin A\cdot\cos B-\cos A\cdot\sin B\)
    \item \(\cos(A+B)=\cos A\cdot\cos B-\sin A\sin B\)
    \item \(\cos(A-B)=\cos A\cdot\cos B+\sin A\cdot\sin B\)
    \item \(\displaystyle \tan(A+B)=\frac{\tan A+\tan B}{1-\tan A\tan B}\)
    \item \(\displaystyle \tan(A-B)=\frac{\tan A-\tan B}{1+\tan A\cdot\tan B}\)
    \item \(\displaystyle \cot(A+B)=\frac{\cot B\cdot\cot A-1}{\cot B+\cot A}\)
    \item \(\displaystyle \cot(A-B)=\frac{\cot B\cdot\cot A+1}{\cot B-\cot A}\)
\end{itemize}

\begin{itemize}
    \item \(\text{odd }F(x)\Rightarrow F(-x)=-F(x)\)
    \item \(\text{Even }F(x)\Rightarrow F(-x)=+F(x)\)
\end{itemize}

\subsection*{*General Solution of trigonometry}

\begin{itemize}
    \item \(\sin x=0\Rightarrow x=n\pi\)
    \item \(\displaystyle
    \cos x=0\Rightarrow x=(2n+1)\frac{\pi}{2}\)
    \item \(\tan x=0\Rightarrow x=n\pi\)
    \item \(\sin x=\sin y\Rightarrow x=n\pi+(-1)^{n}y\)
    \item \(\cos x=\cos y\Rightarrow x=2n\pi\pm y\)
    \item \(\tan x=\tan y\Rightarrow x=n\pi+y\)
\end{itemize}

\[
\left.
\begin{aligned}
\sin^{2}x&=\sin^{2}y\\
\cos^{2}x&=\cos^{2}y\\
\tan^{2}x&=\tan^{2}y
\end{aligned}
\right\}
\quad\Rightarrow\quad x=n\pi\pm y
\]

\[
\boxed{n\in Z}
