\begin{align*}
|TX^R|_{L^\infty_W L^\infty_{[0,T]} L^2_x} &\le |X_0|_{L^\infty_W L^2_x} + \left| \int_0^t S_\beta(t,s) \Theta_R(X^R)(s) |X^R(s)|^{2\sigma} X^R(s) ds \right|_{L^\infty_W L^\infty_{[0,T]} L^2_x} \\
&\le |X_0|_{L^\infty_W L^2_x} + c_{r,p} T^{\tilde{\beta}} |\Theta_R(X^R)| X^R|^{2\sigma+1}|_{L^\gamma_W L^\gamma_{[0,T]} L^p_x} \\
&= |X_0|_{L^\infty_W L^2_x} + c_{r,p} T^{\tilde{\beta}} \left| \left[ \int_0^T \Theta_R(X^R)^\gamma(t) \left( \int |X^R(t)|^{(2\sigma+1)p'} dx \right)^{\frac{\gamma}{p'}} dt \right]^{\frac{1}{\gamma}} \right|_{L^\gamma_W}
\end{align*}

Since \(p' = \frac{2\sigma+2}{2\sigma+1}\), the right hand side equals
\[
= |X_0|_{L^\infty_W L^2_x} + c_{r,p} T^{\tilde{\beta}} \left| \left( \int_0^T \Theta_R(X^R)^\gamma(t) |X^R(t)|_{L^\xi_x}^{(2\sigma+1)\gamma'} dt \right)^{\frac{1}{\gamma}} \right|_{L^\gamma_W}
\]

Since \(\gamma \ge p = 2\sigma+2\), we can apply Hölder's inequality to get
\[
\le |X_0|_{L^\infty_W L^2_x} + c_{r,p} T^{\tilde{\beta}} \left| \left( \int_0^T [\Theta_R(X^R)(t)]^{\frac{\gamma}{2(\sigma+1)}} |X^R(t)|_{L^\xi_x}^{\frac{\gamma}{2(\sigma+1)}} dt \right)^{\frac{2\sigma+1}{\gamma}} \right|_{L^\gamma_W}^{\frac{\gamma-\gamma'(2\sigma+1)}{\gamma\gamma'}}
\]

Letting \(\tilde{t}(w) := \inf\{s : |X^R(w)|_{L^\xi_x([0,s]; L^\xi_x)} \ge 2R\}\), from definition of \(\Theta_R\) in (??), we have
\[
= |X_0|_{L^\infty_W L^2_x} + c_{r,p} T^{\tilde{\beta}} \left| \left( \int_0^{\tilde{t}} [\Theta_R(X^R)(t)]^{\frac{\gamma}{2(\sigma+1)}} |X^R(t)|_{L^\xi_x}^{\frac{\gamma}{2(\sigma+1)}} dt \right)^{\frac{2\sigma+1}{\gamma}} \right|_{L^\gamma_W}
\]
where \(\tilde{\beta} = \bar{\beta} + \frac{\gamma-\gamma'(2\sigma+1)}{\gamma\gamma'} > 0\). Furthermore,
\begin{align*}
&\le |X_0|_{L^\infty_W L^2_x} + c_{r,p} T^{\tilde{\beta}} \left| \left( \int_0^{\tilde{t}} |X^R(t)|_{L^\xi_x}^{\frac{\gamma}{2(\sigma+1)}} dt \right)^{\frac{2\sigma+1}{\gamma}} \right|_{L^\gamma_W} \\
&\le |X_0|_{L^\infty_W L^2_x} + c_{r,p} T^{\tilde{\beta}} (2R)^{2\sigma} |X^R|_{L^\infty_W L^\infty_{[0,T]} L^\xi_x}.
\end{align*}
