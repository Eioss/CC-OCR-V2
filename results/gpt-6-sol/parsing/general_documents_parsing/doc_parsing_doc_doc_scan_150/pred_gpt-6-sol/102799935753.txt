\[
\begin{aligned}
\sin C
&=\frac{\text{side opposite to angle }C}{\text{hypotenuse}}
=\frac{p}{h}\\[6pt]
\cos C
&=\frac{\text{side adjacent to angle }C}{\text{hypotenuse}}
=\frac{b}{h}\\[6pt]
\tan C
&=\frac{\text{side opposite to angle }C}{\text{side adjacent to angle }C}
=\frac{p}{b}
\end{aligned}
\]

\[
\operatorname{cosec}\theta=\frac{1}{\sin\theta};
\quad
\sec\theta=\frac{1}{\cos\theta};
\quad
\cot\theta=\frac{1}{\tan\theta};
\quad
\tan\theta=\frac{\sin\theta}{\cos\theta}
\]
