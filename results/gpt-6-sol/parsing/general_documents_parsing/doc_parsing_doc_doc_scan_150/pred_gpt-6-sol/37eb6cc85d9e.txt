\begin{center}
\textbf{Statistical Analysis and Probability}
\end{center}

\noindent\textbf{Mean}
\[
\bar{x}=\frac{1}{n}\sum_{i=1}^{n}x_i
\]

\noindent\textbf{Standard Deviation*}
\[
s=\sqrt{\frac{\sum(x_i-\bar{x})^2}{n-1}}
\]

\noindent\textbf{Standard Error of the Mean*}
\[
SE_{\bar{x}}=\frac{s}{\sqrt{n}}
\]

\noindent\textbf{Chi-Square}
\[
\chi^2=\sum\frac{(o-e)^2}{e}
\]

\[
\begin{aligned}
\bar{x}&=\text{sample mean}\\
n&=\text{size of the sample}\\
s&=\text{sample standard deviation (i.e., the sample-based}\\
 &\quad\text{estimate of the standard deviation of the population)}\\
o&=\text{observed results}\\
e&=\text{expected results}
\end{aligned}
\]

\noindent Degrees of freedom are equal to the number of distinct possible outcomes minus one.

\medskip
\noindent\textbf{Laws of Probability}

\noindent If A and B are mutually exclusive, then:
\[
P(\text{A or B})=P(\text{A})+P(\text{B})
\]
\noindent If A and B are independent, then:
\[
P(\text{A and B})=P(\text{A})\times P(\text{B})
\]

\medskip
\noindent\textbf{Hardy-Weinberg Equations}
\[
\begin{aligned}
p^2+2pq+q^2&=1
&\qquad p&=\text{frequency of the dominant allele}\\
&&&\quad\text{in a population}\\
p+q&=1
&\qquad q&=\text{frequency of the recessive allele}\\
&&&\quad\text{in a population}
\end{aligned}
\]

\noindent Mode = value that occurs most frequently in a data set

\noindent Median = middle value that separates the greater and lesser halves of a data set

\noindent Mean = sum of all data points divided by number of data points

\noindent Range = value obtained by subtracting the smallest observation (sample minimum) from the greatest (sample maximum)
