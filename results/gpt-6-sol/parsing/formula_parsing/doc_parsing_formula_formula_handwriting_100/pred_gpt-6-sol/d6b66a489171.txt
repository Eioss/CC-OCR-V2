\[
\begin{aligned}
\Sigma_1
&=\frac{(N-N_{\mathrm{de}})!}{N!}
\sum_{\alpha_1}
\sum_{\alpha_2\ne\alpha_1}\cdots
\sum_{\alpha_{N_{\mathrm{de}}}\ne\alpha_1,\ldots,\alpha_{N_{\mathrm{de}}-1}}
\sum_{i\in D}\sum_{j\ne i}\Gamma_{ij}\Gamma_{ji}\\
&=\frac{(N-N_{\mathrm{de}})!}{N!}
\sum_{\alpha_1}
\sum_{\alpha_2\ne\alpha_1}\cdots
\sum_{\alpha_{N_{\mathrm{de}}}\ne\alpha_1,\ldots,\alpha_{N_{\mathrm{de}}-1}}
\left(
\sum_{j\ne\alpha_1}\Gamma_{\alpha_1j}\Gamma_{j\alpha_1}
+\cdots+
\sum_{j\ne\alpha_{N_{\mathrm{de}}}}
\Gamma_{\alpha_{N_{\mathrm{de}}}j}\Gamma_{j\alpha_{N_{\mathrm{de}}}}
\right)\\
&=N_{\mathrm{de}}\frac{(N-N_{\mathrm{de}})!}{N!}
\sum_{\alpha_1}\sum_{j\ne\alpha_1}
\Gamma_{\alpha_1j}\Gamma_{j\alpha_1}
\left(
\sum_{\alpha_2\ne\alpha_1}\cdots
\sum_{\alpha_{N_{\mathrm{de}}}\ne\alpha_1,\ldots,\alpha_{N_{\mathrm{de}}-1}}1
\right)\\
&=\frac{N_{\mathrm{de}}}{N}\sum_i\sum_{j\ne i}\Gamma_{ij}\Gamma_{ji},
\\[1em]
\Sigma_2
&=\frac{(N-N_{\mathrm{de}})!}{N!}
\sum_{\alpha_1}
\sum_{\alpha_2\ne\alpha_1}\cdots
\sum_{\alpha_{N_{\mathrm{de}}}\ne\alpha_1,\ldots,\alpha_{N_{\mathrm{de}}-1}}
\sum_{i\in D}\sum_{\substack{j\in D\\j\ne i}}\Gamma_{ij}\Gamma_{ji}\\
&=\frac{(N-N_{\mathrm{de}})!}{N!}
\sum_{\alpha_1}
\sum_{\alpha_2\ne\alpha_1}\cdots
\sum_{\alpha_{N_{\mathrm{de}}}\ne\alpha_1,\ldots,\alpha_{N_{\mathrm{de}}-1}}
\left(
\Gamma_{\alpha_1\alpha_2}\Gamma_{\alpha_2\alpha_1}
+\cdots+
\Gamma_{\alpha_1\alpha_{N_{\mathrm{de}}}}
\Gamma_{\alpha_{N_{\mathrm{de}}}\alpha_1}
+\Gamma_{\alpha_2\alpha_1}\Gamma_{\alpha_1\alpha_2}
+\cdots
\right)\\
&=N_{\mathrm{de}}(N_{\mathrm{de}}-1)
\frac{(N-N_{\mathrm{de}})!}{N!}
\sum_{\alpha_1}\sum_{\alpha_2\ne\alpha_1}
\Gamma_{\alpha_1\alpha_2}\Gamma_{\alpha_2\alpha_1}
\left(
\sum_{\alpha_3\ne\alpha_1,\alpha_2}\cdots
\sum_{\alpha_{N_{\mathrm{de}}}\ne\alpha_1,\ldots,\alpha_{N_{\mathrm{de}}-1}}1
\right)\\
&=\frac{N_{\mathrm{de}}(N_{\mathrm{de}}-1)}{N(N-1)}
\sum_i\sum_{j\ne i}\Gamma_{ij}\Gamma_{ji}.
\end{aligned}
