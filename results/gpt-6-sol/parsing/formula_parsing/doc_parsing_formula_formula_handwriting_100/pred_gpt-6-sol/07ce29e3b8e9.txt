\[
\begin{aligned}
&\left|\int_{t_n}^{t}\int_{-\infty}^{\infty}
\frac{u^{2}(s,\tau)}{2}
\left(G_x(x,t;s,\tau)h_x(s)+G_{xx}(x,t;s,\tau)h(s)\right)
\,ds\,d\tau\right|\\
&\quad\leq
\frac{\|u_x\|_\infty+1}{2}\|h_x\|_\infty
\int_{t_n}^{t}\frac{1}{\sqrt{\pi(t-\tau)}}\,d\tau\\
&\qquad+
\left|\int_{t_n}^{t}\int_{-\infty}^{\infty}
\left(
u^{2}(x+\lambda\sqrt{t-\tau},\tau)
h_x(x+\lambda\sqrt{t-\tau})
-u^{2}(x,\tau)h_x(x)
\right)
\frac{\lambda^{2}-\tfrac12}{\sqrt{\pi(t-\tau)}}
e^{-\lambda^{2}}\,d\lambda\,d\tau\right|\\
&\quad\leq
C(\|u\|_{\infty,\alpha},\alpha,\beta)
\int_{t_n}^{t}\int_{-\infty}^{\infty}
\frac{(\sqrt{t-\tau}\,|\lambda|)^\beta}{\sqrt{\tau}}\,
\frac{|\lambda^{2}-\tfrac12|}{\sqrt{t-\tau}}\,
e^{-\lambda^{2}}\,d\lambda\,d\tau\\
&\quad\leq
C(\|u\|_{\infty,\alpha},\alpha,\beta)(2\pi)^{-1/2}
\int_{t_n}^{t}
\frac{1}{\sqrt{\tau}\,(t-\tau)^{(1-\beta)/2}}\,d\tau .
\end{aligned}
