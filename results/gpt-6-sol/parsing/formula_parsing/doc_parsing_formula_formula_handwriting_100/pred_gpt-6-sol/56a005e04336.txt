\begin{aligned}
&\int_0^\infty r^{N-1}\sup_{0\le s\le r}
\int_{S^{N-1}}|u(s\sigma)|^p\,d\sigma\,dr\\
&=\int_0^\infty r^{N-1}\sup_{0\le s\le r}
\int_{S^{N-1}}\frac{|u(s\sigma)|}{r}\,
r|u(s\sigma)|^{p-1}\,d\sigma\,dr\\
&\le \int_0^\infty r^{N-1}\sup_{0\le s\le r}
\left(\int_{S^{N-1}}\frac{|u(s\sigma)|^p}{r^p}\,d\sigma\right)^{1/p}

&\le \int_0^\infty r^{N-1}
\sup_{0\le s\le r}\left(\int_{S^{N-1}}\frac{|u(s\sigma)|^p}{r^p}\,d\sigma\right)^{1/p}
\sup_{0\le s\le r}\left(\int_{S^{N-1}}r^{\frac{p}{p-1}}|u(s\sigma)|^p\,d\sigma\right)^{\frac{p-1}{p}}dr\\
&=\int_0^\infty
\left\{r^{\frac{N-1}{p}}\sup_{0\le s\le r}
\left(\int_{S^{N-1}}\frac{|u(s\sigma)|^p}{r^p}\,d\sigma\right)^{1/p}\right\}
\left\{r^{\frac{(N-1)(p-1)}{p}}\sup_{0\le s\le r}
\left(\int_{S^{N-1}}r^{\frac{p}{p-1}}|u(s\sigma)|^p\,d\sigma\right)^{\frac{p-1}{p}}\right\}dr\\
&\le
\left(\int_0^\infty r^{N-1}\sup_{0\le s\le r}
\int_{S^{N-1}}\frac{|u(s\sigma)|^p}{r^p}\,d\sigma\,dr\right)^{1/p}
\left(\int_0^\infty r^{N-1}\sup_{0\le s\le r}
