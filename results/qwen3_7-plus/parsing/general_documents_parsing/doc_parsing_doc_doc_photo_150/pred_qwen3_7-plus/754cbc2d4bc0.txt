\section*{TRIGONOMETRY}

\begin{itemize}
    \item $\sin(A+B) = \sin A \cdot \cos B + \cos A \cdot \sin B$
    \item $\sin(A-B) = \sin A \cdot \cos B - \cos A \cdot \sin B$
    \item $\cos(A+B) = \cos A \cdot \cos B - \sin A \sin B$
    \item $\cos(A-B) = \cos A \cdot \cos B + \sin A \cdot \sin B$
    \item $\tan(A+B) = \frac{\tan A + \tan B}{1 - \tan A \tan B}$
    \item $\tan(A-B) = \frac{\tan A - \tan B}{1 + \tan A \cdot \tan B}$
    \item $\cot(A+B) = \frac{\cot B \cot A - 1}{\cot B + \cot A}$
    \item $\cot(A-B) = \frac{\cot B \cdot \cot A + 1}{\cot B - \cot A}$
\end{itemize}

\begin{itemize}
    \item odd $F(x) \Rightarrow F(-x) = -F(x)$
    \item Even $f(x) \Rightarrow f(-x) = +F(x)$
\end{itemize}

\subsection*{*General Solution of trigonometry}

\begin{itemize}
    \item $\sin x = 0 \Rightarrow x = n\pi$
    \item $\cos x = 0 \Rightarrow x = (2n+1)\frac{\pi}{2}$
    \item $\tan x = 0 \Rightarrow x = n\pi$
    \item $\sin x = \sin y \Rightarrow x = n\pi + (-1)^n y$
    \item $\cos x = \cos y \Rightarrow x = 2n\pi \pm y$
    \item $\tan x = \tan y \Rightarrow x = n\pi + y$
\end{itemize}

\begin{itemize}
    \item $\sin 2x = 2 \sin x \cdot \cos x = \frac{2 \tan x}{1 + \tan^2 x}$
    \item $\cos 2x = \cos^2 x - \sin^2 x = 2 \cos^2 x - 1$
    \item $\qquad = 1 - 2 \sin^2 x = \frac{1 - \tan^2 x}{1 + \tan^2 x}$
    \item $\tan 2x = \frac{2 \tan x}{1 - \tan^2 x}$
    \item $\sin 3x = 3 \sin x - 4 \sin^3 x$
    \item $\cos 3x = 4 \cos^3 x - 3 \cos x$
    \item $\tan 3x = \frac{3 \tan x - \tan^3 x}{1 - 3 \tan^2 x}$
\end{itemize}

\begin{itemize}
    \item $a \sin x + b \cos x$
    \item Max. Value = $+\sqrt{a^2+b^2}$
    \item Min. Value = $-\sqrt{a^2+b^2}$
\end{itemize}

\begin{itemize}
    \item $\sin^2 x = \sin^2 y$
    \item $\cos^2 x = \cos^2 y$
    \item $\tan^2 x = \tan^2 y$
\end{itemize}
$\Rightarrow x = n\pi \pm y$

$n \in \mathbb{Z}$
