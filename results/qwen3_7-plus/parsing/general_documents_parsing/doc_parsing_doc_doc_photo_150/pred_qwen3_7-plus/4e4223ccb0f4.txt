\noindent \textbf{1. Express each number as a product of its prime factors:}

\begin{enumerate}[(i)]
    \item 140 \\

\end{enumerate}

\noindent \textbf{2. Find the LCM and HCF of the following pairs of integers and verify that LCM $\times$ HCF = Product of the two numbers}

\begin{enumerate}[(i)]
    \item 26 and 91 \\
    Prime factors of $26 = 2 \times 13$ \\
    Prime factors of $91 = 7 \times 13$ \\
    HCF of 26 and $91 = 13$ \\
    LCM of 26 and $91 = 2 \times 7 \times 13$ \\
    \phantom{LCM of 26 and } $= 182$ \\[1em]
    Product of two numbers $= 26 \times 91 = 2366$ \\
    LCM $\times$ HCF $= 182 \times 13 = 2366$ \\
    So, Product of two numbers = LCM $\times$ HCF
\end{enumerate}
