    \begin{tabular*}{\textwidth}{@{}l@{\extracolsep{\fill}}r@{}}
        a) one-to-one & b) onto \\
        c) bijection & d) cannot be defined
    \end{tabular*}

    \item The solution set of the following inequality $|x-1| \ge |x-3|$ is
    \begin{tabular*}{\textwidth}{@{}l@{\extracolsep{\fill}}r@{}}
        a) $[0, 2]$ & b) $[2, \infty)$ \\
        c) $(0, 2)$ & d) $(-\infty, 2)$
    \end{tabular*}

    \item If 3 is the logarithm of 343, then the base is
    \begin{tabular*}{\textwidth}{@{}l@{\extracolsep{\fill}}r@{}}
        a) 5 & b) 7 \\
        c) 6 & d) 9
    \end{tabular*}

    \item In a $\Delta ABC$, $\tan\left(\frac{A}{2}\right) =$
    \begin{tabular*}{\textwidth}{@{}l@{\extracolsep{\fill}}r@{}}
        a) $\sqrt{\frac{(s-b)(s-c)}{bc}}$ & b) $\sqrt{\frac{s(s-a)}{bc}}$ \\
        c) $\sqrt{\frac{(s-b)(s-c)}{s(s-a)}}$ & d) $\sqrt{s(s-a)(s-b)(s-c)}$
    \end{tabular*}

    \item There are 10 points in a plane and 4 of them are collinear. The number of straight lines joining any two points is
    \begin{tabular*}{\textwidth}{@{}l@{\extracolsep{\fill}}r@{}}
        a) 45 & b) 40 \\
        c) 39 & d) 38
    \end{tabular*}

    \item The $n^{\text{th}}$ term of the sequence 1, 2, 4, 7, 11, ......... is
    \begin{tabular*}{\textwidth}{@{}l@{\extracolsep{\fill}}r@{}}
        a) $n^3+3n^2+2n$ & b) $n^3-3n^2+3n$ \\
        c) $\frac{n(n+1)(n+2)}{3}$ & d) $\frac{n^2-n+2}{2}$
    \end{tabular*}

    \begin{tabular*}{\textwidth}{@{}l@{\extracolsep{\fill}}r@{}}

    \end{tabular*}
\end{enumerate}
