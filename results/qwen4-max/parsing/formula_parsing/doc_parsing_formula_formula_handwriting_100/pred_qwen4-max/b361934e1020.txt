\begin{align*}
(\ref{eq:2.1}) &\lesssim \int_{|y|<1} |x-y|^{-d+\alpha-1} \frac{\delta_B(y)^{\alpha/2}}{\gamma_B(x,y)^{\alpha}} |y-w|^{-d+\alpha-\beta} \frac{\delta_B(w)^{\alpha/2}}{\gamma_B(y,w)^{\alpha/2}} \left( 1 \vee \frac{|y-w|^{\beta-\alpha/2}}{\delta_B(y)^{\beta-\alpha/2}} \right) dy \\
&\le \int_{|y|<1, |y-w|\ge \delta_B(y)} \frac{\delta_B(w)^{\alpha/2}}{|x-y|^{d-\alpha+1}|w-y|^{d-\alpha+\beta}} \frac{\delta_B(w)^{\alpha/2}}{\gamma_B(y,w)^{\alpha/2}} dy \\
&\quad + \int_{|y|<1, |y-w|<\delta_B(y)} \frac{\delta_B(w)^{\alpha/2}}{|x-y|^{d-\alpha+1}|w-y|^{d-\alpha+\beta}} \frac{\delta_B(y)^{\alpha/2}}{\gamma_B(x,y)^{\alpha}\gamma_B(y,w)^{\alpha/2}} dy \\
&\lesssim \delta_B(w)^{\alpha/2} \int_{|y|<1, |y-w|\ge \delta_B(y)} |x-y|^{-d+\alpha-1} |y-w|^{-d+\alpha-\beta} dy \\
&\quad + \delta_B(w)^{\alpha/2} \int_{|y|<1, |y-w|<\delta_B(y)} |x-y|^{-d+\alpha-1} |y-w|^{-d+\alpha-\beta} dy \\
&= \delta_B(w)^{\alpha/2} \int_{|y|<1} |x-y|^{-d+\alpha-1} |y-w|^{-d+\alpha-\beta} dy \\
&\lesssim \delta_B(w)^{\alpha/2} |x-w|^{-d+(\alpha-1)\wedge(\alpha-\beta)} \int_{|y|<1} (|x-y|^{(\alpha-1)-(\alpha-1)\wedge(\alpha-\beta)} + |y-w|^{\alpha-\beta-(\alpha-1)\wedge(\alpha-\beta)}) dy \\
&\lesssim |x-w|^{-d+(\alpha-1)\wedge(\alpha-\beta)} \delta_B(w)^{\alpha/2}
\end{align*}
