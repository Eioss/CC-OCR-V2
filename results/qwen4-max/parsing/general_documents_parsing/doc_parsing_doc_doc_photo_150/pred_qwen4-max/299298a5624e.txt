\textbf{5) Compound Angles :}
\begin{align*}
i) \sin(A+B) &= \sin A \cos B + \cos A \sin B \\
ii) \sin(A-B) &= \sin A \cos B - \cos A \sin B \\
iii) \cos(A+B) &= \cos A \cos B - \sin A \sin B \\
iv) \cos(A-B) &= \cos A \cos B + \sin A \sin B \\
v) \tan(A+B) &= \frac{\tan A + \tan B}{1 - \tan A \tan B} \\
vi) \tan(A-B) &= \frac{\tan A - \tan B}{1 + \tan A \tan B}
\end{align*}

\textbf{6) Multiple Angle and Sub Multiple Angle :}
\begin{align*}
i) \sin 2A &= 2 \sin A \cos A & i) \sin A &= 2 \sin\left(\frac{A}{2}\right) \cos\left(\frac{A}{2}\right) \\
&= \frac{2 \tan A}{1 + \tan^2 A} & ii) \cos A &= \cos^2\left(\frac{A}{2}\right) - \sin^2\left(\frac{A}{2}\right) \\
ii) \cos 2A &= \cos^2 A - \sin^2 A & iii) \cos A &= 1 - 2 \sin^2\left(\frac{A}{2}\right) \\
&= 2 \cos^2 A - 1 & iv) \cos A &= 2 \cos^2\left(\frac{A}{2}\right) - 1 \\
&= 1 - 2 \sin^2 A & v) \cos A &= \frac{1 - \tan^2(A/2)}{1 + \tan^2(A/2)} \\
&= \frac{1 - \tan^2 A}{1 + \tan^2 A} & vi) \sin A &= \frac{2 \tan(A/2)}{1 + \tan^2(A/2)} \\
iii) \tan 2A &= \frac{2 \tan A}{1 - \tan^2 A} & vii) \tan A &= \frac{2 \tan(A/2)}{1 - \tan^2(A/2)} \\
& & viii) 1 - \cos A &= 2 \sin^2\left(\frac{A}{2}\right) \\
& & 1 + \cos A &= 2 \cos^2\left(\frac{A}{2}\right)
\end{align*}

\textbf{7) Triple Angle Formulae :-}
\begin{align*}
i) \sin 3\theta &= 3 \sin \theta - 4 \sin^3 \theta \\
ii) \cos 3\theta &= 4 \cos^3 \theta - 3 \cos \theta \\
iii) \tan 3\theta &= \frac{3 \tan \theta - \tan^3 \theta}{1 - 3 \tan^2 \theta} \\
iv) \cos^3 \theta &= \frac{\cos 3\theta + 3 \cos \theta}{4} \\
v) \sin^3 \theta &= \frac{3 \sin \theta - \sin 3\theta}{4}
\end{align*}

\textbf{8) Factorization Formulae :-}
\begin{align*}
i. \sin C + \sin D &= 2 \sin\left(\frac{C+D}{2}\right) \cos\left(\frac{C-D}{2}\right) \\
ii. \sin C - \sin D &= 2 \sin\left(\frac{C-D}{2}\right) \cos\left(\frac{C+D}{2}\right) \\
iii. \cos C + \cos D &= 2 \cos\left(\frac{C+D}{2}\right) \cos\left(\frac{C-D}{2}\right) \\
iv. \cos C - \cos D &= 2 \sin\left(\frac{C+D}{2}\right) \sin\left(\frac{C-D}{2}\right) \text{ if } C>D \\
&= 2 \sin\left(\frac{C+D}{2}\right) \sin\left(\frac{D-C}{2}\right) \text{ if } D>C
\end{align*}
