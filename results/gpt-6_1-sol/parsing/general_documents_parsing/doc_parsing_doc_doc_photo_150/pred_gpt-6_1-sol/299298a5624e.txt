\subsection*{5) Compound Angles}

\[
\begin{aligned}
\text{i)}\quad
\sin(A+B)&=\sin A\cos B+\cos A\sin B\\
\text{ii)}\quad
\sin(A-B)&=\sin A\cos B-\cos A\sin B\\
\text{iii)}\quad
\cos(A-B)&=\cos A\cos B+\sin A\sin B\\
\text{iv)}\quad
\cos(A+B)&=\cos A\cos B-\sin A\sin B\\
\text{v)}\quad
\tan(A+B)&=\frac{\tan A+\tan B}{1-\tan A\tan B}\\
\text{vi)}\quad
\tan(A-B)&=\frac{\tan A-\tan B}{1+\tan A\tan B}
\end{aligned}
\]

\subsection*{7) Triple Angle Formulae}

\[
\begin{aligned}
\text{i)}\quad
\sin3\theta&=3\sin\theta-4\sin^3\theta\\
\text{ii)}\quad
\cos3\theta&=4\cos^3\theta-3\cos\theta\\
\text{iii)}\quad
\tan3\theta&=\frac{3\tan\theta-\tan^3\theta}
                  {1-3\tan^2\theta}\\
\text{iv)}\quad
\cos^3\theta&=\frac{\cos3\theta+3\cos\theta}{4}\\
\text{v)}\quad
\sin^3\theta&=\frac{3\sin\theta-\sin3\theta}{4}
\end{aligned}
\]

\subsection*{8) Factorization Formulae}

\[
\begin{aligned}
\text{i)}\quad
\sin C+\sin D
&=2\sin\left(\frac{C+D}{2}\right)
    \cos\left(\frac{C-D}{2}\right)\\
\text{ii)}\quad
\sin C-\sin D
&=2\sin\left(\frac{C-D}{2}\right)
    \cos\left(\frac{C+D}{2}\right)\\
\text{iii)}\quad
\cos C+\cos D
&=2\cos\left(\frac{C+D}{2}\right)
    \cos\left(\frac{C-D}{2}\right)\\
\text{iv)}\quad
\cos C-\cos D
&=2\sin\left(\frac{C+D}{2}\right)
    \sin\left(\frac{C-D}{2}\right)
    \quad\text{if }C>D\\
&=2\sin\left(\frac{C+D}{2}\right)
    \sin\left(\frac{D-C}{2}\right)
    \quad\text{if }D>C
\end{aligned}
\]
