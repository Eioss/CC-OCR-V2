\[
\underline{\text{TRIGONOMETRY}}
\]

\[
\begin{aligned}
\sin(A+B)&=\sin A\cdot\cos B+\cos A\cdot\sin B\\
\sin(A-B)&=\sin A\cdot\cos B-\cos A\cdot\sin B\\
\cos(A+B)&=\cos A\cdot\cos B-\sin A\cdot\sin B\\
\cos(A-B)&=\cos A\cdot\cos B+\sin A\cdot\sin B\\[4pt]
\tan(A+B)&=\frac{\tan A+\tan B}{1-\tan A\tan B}\\[4pt]
\tan(A-B)&=\frac{\tan A-\tan B}{1+\tan A\tan B}\\[4pt]
\cot(A+B)&=\frac{\cot B\cot A-1}{\cot B+\cot A}\\[4pt]
\cot(A-B)&=\frac{\cot B\cot A+1}{\cot B-\cot A}
\end{aligned}
\]

\[
\begin{aligned}
\text{odd }f(x)&\Rightarrow f(-x)=-f(x)\\
\text{Even }f(x)&\Rightarrow f(-x)=f(x)
\end{aligned}
\]

\[
\begin{array}{l|l}
\cos(-x)=\cos x & \sec(-x)=\sec x\\
\sin(-x)=-\sin x & \operatorname{cosec}(-x)=-\operatorname{cosec}x\\
\tan(-x)=-\tan x & \cot(-x)=-\cot x
\end{array}
\]

\[
\begin{aligned}
2\sin A\cdot\cos B&=\sin(A+B)+\sin(A-B)\\
2\cos A\cdot\sin B&=\sin(A+B)-\sin(A-B)\\
2\cos A\cdot\cos B&=\cos(A+B)+\cos(A-B)\\
-2\sin A\cdot\sin B&=\cos(A+B)-\cos(A-B)\\
\sin(A+B)\cdot\sin(A-B)&=\sin^2 A-\sin^2 B\\
\cos(A+B)\cdot\cos(A-B)&=\cos^2 A-\sin^2 B
\end{aligned}
\]

\[
\begin{aligned}
\sin C+\sin D
&=2\sin\left(\frac{C+D}{2}\right)\cdot
\cos\left(\frac{C-D}{2}\right)\\[4pt]
\sin C-\sin D
&=2\cos\left(\frac{C+D}{2}\right)\cdot
\sin\left(\frac{C-D}{2}\right)\\[4pt]
\cos C+\cos D
&=2\cos\left(\frac{C+D}{2}\right)\cdot
\cos\left(\frac{C-D}{2}\right)\\[4pt]
\cos C-\cos D
&=-2\sin\left(\frac{C+D}{2}\right)\cdot
\sin\left(\frac{C-D}{2}\right)\\
&=2\sin\left(\frac{C+D}{2}\right)\cdot
\sin\left(\frac{D-C}{2}\right)
\end{aligned}
\]

\[
\boxed{
\begin{array}{l|l}
\sin^2\theta+\cos^2\theta=1
&1+\cos2x=2\cos^2x\\[3pt]
\sec^2\theta-\tan^2\theta=1
&1-\cos2x=2\sin^2x\\[3pt]
\operatorname{cosec}^2\theta-\cot^2\theta=1
&\displaystyle\tan x=\frac{1-\cos2x}{\sin2x}
\end{array}
}
\]

\[
\begin{aligned}
\sin2x
&=2\sin x\cdot\cos x
=\frac{2\tan x}{1+\tan^2x}\\[4pt]
\cos2x
&=\cos^2x-\sin^2x
=2\cos^2x-1\\
&=1-2\sin^2x
=\frac{1-\tan^2x}{1+\tan^2x}\\[4pt]
\tan2x
&=\frac{2\tan x}{1-\tan^2x}\\[6pt]
\sin3x
&=3\sin x-4\sin^3x\\
\cos3x
&=4\cos^3x-3\cos x\\[4pt]
\tan3x
&=\frac{3\tan x-\tan^3x}{1-3\tan^2x}
\end{aligned}
\]

\[
\underline{\text{General Solution of trigonometry}}
\]

\[
\begin{aligned}
\sin x=0&\Rightarrow x=n\pi\\
\cos x=0&\Rightarrow x=\frac{(2n+1)\pi}{2}\\
\tan x=0&\Rightarrow x=n\pi\\[4pt]
\sin x=\sin y&\Rightarrow x=n\pi+(-1)^n\cdot y\\
\cos x=\cos y&\Rightarrow x=2n\pi\pm y\\
\tan x=\tan y&\Rightarrow x=n\pi+y
\end{aligned}
\]

\[
\left.
\begin{aligned}
\sin^2x&=\sin^2y\\
\cos^2x&=\cos^2y\\
\tan^2x&=\tan^2y
\end{aligned}
\right\}
\Rightarrow x=n\pi\pm y
\qquad
\boxed{n\in\mathbb Z}
\]

\end{aligned}
