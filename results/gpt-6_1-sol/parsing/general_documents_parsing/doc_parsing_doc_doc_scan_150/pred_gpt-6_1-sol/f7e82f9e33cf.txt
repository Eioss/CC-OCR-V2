\textbf{formulas for discrete-time LTI signals and systems}

\medskip
\begin{tabular}{@{}ll@{}}
\hline
name & formula \\
\hline
area under impulse
& $\displaystyle \sum_n \delta(n)=1$ \\[10pt]

multiplication by impulse
& $f(n)\delta(n)=f(0)\delta(n)$ \\[8pt]

$\ldots$ by shifted impulse
& $f(n)\delta(n-n_0)=f(n_0)\delta(n-n_0)$ \\[8pt]

convolution
& $\displaystyle f(n)*g(n)=\sum_k f(k)\,g(n-k)$ \\[10pt]

$\ldots$ with an impulse
& $f(n)*\delta(n)=f(n)$ \\[8pt]

$\ldots$ with a shifted impulse
& $f(n)*\delta(n-n_0)=f(n-n_0)$ \\[8pt]

transfer function
& $\displaystyle H(z)=\sum_n h(n)\,z^{-n}$ \\[10pt]

frequency response
& $\displaystyle H^f(\omega)=\sum_n h(n)\,e^{-j\omega n}$ \\[8pt]

$\ldots$ their connection
& $\begin{gathered}
H^f(\omega)=H(e^{j\omega})\\
\text{provided unit circle}\subset\text{ROC}
\end{gathered}$ \\[10pt]

\end{tabular}

\textbf{formulas for continuous-time LTI signals and systems}

\medskip
\begin{tabular}{@{}ll@{}}
\hline
name & formula \\
\hline
area under impulse
& $\displaystyle \int \delta(t)\,dt=1$ \\[10pt]

multiplication by impulse
& $f(t)\delta(t)=f(0)\delta(t)$ \\[8pt]

$\ldots$ by shifted impulse
& $f(t)\delta(t-t_0)=f(t_0)\delta(t-t_0)$ \\[8pt]

convolution
& $\displaystyle f(t)*g(t)=\int f(\tau)\,g(t-\tau)\,d\tau$ \\[10pt]

$\ldots$ with an impulse
& $f(t)*\delta(t)=f(t)$ \\[8pt]

$\ldots$ with a shifted impulse
& $f(t)*\delta(t-t_0)=f(t-t_0)$ \\[8pt]

transfer function
& $\displaystyle H(s)=\int h(t)\,e^{-st}\,dt$ \\[10pt]

frequency response
& $\displaystyle H^f(\omega)=\int h(t)\,e^{-j\omega t}\,dt$ \\[8pt]

$\ldots$ their connection
& $\begin{gathered}
H^f(\omega)=H(j\omega)\\
\text{provided }j\omega\text{-axis}\subset\text{ROC}
\end{gathered}$ \\[10pt]
\hline
\end{tabular}
