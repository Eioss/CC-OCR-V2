\textbf{1-d motion, constant \(a\)}
\[
a=\frac{v_f-v_0}{t}
\]
\[
v_{\mathrm{av}}=\frac{v_0+v_f}{2},\qquad
(x_f-x_0)=v_{\mathrm{av}}t
\]
\[
(x_f-x_0)=v_0t+\frac12at^2
=v_ft-\frac12at^2
\]
\[
\frac12v_f^2-\frac12v_0^2=a(x_f-x_0)
\]

\textbf{Projectile Motion}
\[
\text{Range}=\frac{v^2}{g}\sin2\theta,\qquad
\text{Max. height}=\frac{v_0^2}{2g}\sin^2\theta
\]

\textbf{Momentum, Force and Impulse}
\[
p=mv,\quad
F=\frac{\Delta p}{\Delta t}=ma,\quad
I=F\Delta t=\Delta p
\]

\textbf{Work, Energy and Power}
\[
W=\vec F\cdot(\vec r-\vec r_0),\quad
KE=\frac12mv^2,\quad
P=\frac{\Delta E}{\Delta t}
\]

\columnbreak

\textbf{Gravity and circular orbits}
\[
PE=-G\frac{Mm}{r},\quad
\Delta PE=mgh\text{(small }h\text{)},\quad
F=G\frac{Mm}{r^2}=mg
\]
\[
a=\frac{v^2}{r},\qquad
\frac{GM}{4\pi^2}=\frac{R^3}{T^2}
\]

\textbf{Rotational Motion \& Gravity}
\[
v=\omega r=\frac{2\pi r}{T},\quad
\omega=2\pi f=\frac{2\pi}{T},\quad f=1/T
\]
\[
\alpha=\frac{\omega_f-\omega_0}{t}=\frac{a}{r}
\]
\[
L=I\omega=mvr\sin\theta,\quad
(\theta=\text{angle between }\mathbf{v}\text{ and }\mathbf{r})
\]
\[
KE=\frac{L^2}{2I}=\frac12I\omega^2
\]
\[
\tau=rF\sin\theta,\quad I\alpha=\tau,\quad
I_{\mathrm{point}}=mR^2
\]
\[
I_{\mathrm{cyl.shell}}=mR^2,\quad
I_{\mathrm{sphere}}=\frac25mR^2,\quad
I_{\mathrm{solid\ cyl.}}=\frac12mR^2.
\]

\textbf{Gases, liquids and solids}
\[
PV=NRT,\quad
\Delta P=\rho gh,\quad
\left\langle\frac12mv^2\right\rangle=\frac32k_BT
\]
\[
F_{\mathrm{buoyant}}
=\rho_{\mathrm{displaced\ liq.}}
 V_{\mathrm{displaced\ liq.}}g
\]
\[
\text{Stress}=F/A,\quad
\text{Strain}=\Delta L/L,\quad
Y=\text{Stress}/\text{Strain}
\]
\[
\frac{\Delta L}{L}=\frac{F/A}{B}=-\frac{\Delta P}{B}.
\]
\[
P_a+\frac12\rho_av_a^2+\rho_agh_a
=P_b+\frac12\rho_bv_b^2+\rho_bgh_b
\]
\[
\Delta L/L=\alpha\Delta T,\quad
\Delta V/V=3\alpha\Delta T
\]
\[
Q=mC_v\Delta T+mL\text{(if phase trans.)},\quad
\Delta Q/\Delta t=\frac{kA(T_b-T_a)}{x_b-x_a}
\]

\textbf{Radiation}

\textbf{Thermodynamics}

\(\Delta U=Q-W,\quad W=P\Delta V,\quad
\text{ideal gas: }\Delta U=nC_V\Delta T\)

Adiabatic exp: \(pV^\gamma=\mathrm{const},\
TV^{\gamma-1}=\mathrm{const}\)

\(\gamma=C_P/C_V=5/3\text{ for monotonic gas}
=7/5\text{ for diatomic gas}\)

\(Q=T\Delta S,\quad \Delta S>0\)

Engines: \(\epsilon=W/Q_H<(T_H-T_L)/T_H<1\)

Refrigerators and heat pumps:
\(\epsilon=Q_L/W<T_L/(T_H-T_L)\)

\textbf{Simple Harmonic Motion and Waves}

Spring: \(F=-kx,\ PE=(1/2)kx^2,\ \omega=\sqrt{k/m}\)

\(f=\omega/(2\pi),\
x(t)=A\cos(\omega t)+B\sin(\omega t)\)

Pendulum: \(T=2\pi\sqrt{L/g}\)

Waves: \(y(x,t)=A\sin[2\pi(ft-x/\lambda+\delta)],\
v=f\lambda\)

\(I=\mathrm{const}\,A^2f^2,\quad
I_2/I_1=R_1^2/R_2^2\)

Standing waves: \(\lambda_n=2L/n\)

Strings: \(v=\sqrt{T/\mu}\), Solid/Liquid: \(v=\sqrt{B/\rho}\)

Sound: \(I=E/(A\cdot\Delta t)=\mathrm{Power}/A\)

\(I_0=10^{-12}\ \mathrm{W/m^2},\
\text{Intensity in decibels}=10\log_{10}(I/I_0)\)

Beat freq.\(=|f_1-f_2|\), Doppler:
\[
f_{\mathrm{obs}}
=f_{\mathrm{source}}
\frac{v_{\mathrm{sound}}\pm v_{\mathrm{obs}}}
     {v_{\mathrm{sound}}\pm v_{\mathrm{source}}}
\]

Pipes: same at both ends: \(L=\lambda/2,\lambda,3\lambda/2\)

Pipes: open at only one end: \(L=\lambda/4,3\lambda/4\cdots\)
