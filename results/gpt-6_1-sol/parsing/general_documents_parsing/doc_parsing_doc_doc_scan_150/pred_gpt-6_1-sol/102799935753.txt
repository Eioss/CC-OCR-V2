\[
\sin C
= \frac{\textit{side opposite to angle } C}{\textit{hypotenuse}}
= \frac{p}{h}
\]

\[
\cos C
= \frac{\textit{side adjacent to angle } C}{\textit{hypotenuse}}
= \frac{b}{h}
\]

\[
\tan C
= \frac{\textit{side opposite to angle } C}
       {\textit{side adjacent to angle } C}
= \frac{p}{b}
\]

\[
\operatorname{cosec}\theta = \frac{1}{\sin\theta};
\quad
\sec\theta = \frac{1}{\cos\theta};
\quad
\cot\theta = \frac{1}{\tan\theta};
\quad
\tan\theta = \frac{\sin\theta}{\cos\theta}
\]
